<script>
document.addEventListener('DOMContentLoaded', function() {
    // Favorite Logic
    const favoriteBtn = document.getElementById('favoriteProductBtn');
    if (favoriteBtn) {
        favoriteBtn.addEventListener('click', function() {
            this.classList.toggle('active');
            const textSpan = this.querySelector('.btn-text');
            if (this.classList.contains('active')) {
                textSpan.textContent = 'Favorited';
            } else {
                textSpan.textContent = 'Favorite';
            }
        });
    }

    // Voting Logic (Simplified Social Popularity)
    const voteBtns = [document.getElementById('voteUpBtn'), document.getElementById('voteDownBtn')];
    voteBtns.forEach(btn => {
        if (!btn) return;
        btn.addEventListener('click', function() {
            // Visual feedback - allow multiple clicks but don't persist active state after refresh
            const productId = this.getAttribute('data-id');
            const type = this.id === 'voteUpBtn' ? 'up' : 'down';
            const csrfMeta = document.querySelector('meta[name="csrf-token"]');
            const csrfToken = csrfMeta ? csrfMeta.getAttribute('content') : '';

            // Optimistic UI update
            const countSpan = this.querySelector('.count');
            countSpan.textContent = parseInt(countSpan.textContent) + 1;
            this.classList.add('active');

            fetch('https://creativeconor.com/api/vote', {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/x-www-form-urlencoded',
                    'X-Requested-With': 'XMLHttpRequest'
                },
                body: `product_id=${productId}&type=${type}&csrf_token=${csrfToken}`
            })
            .then(response => response.json())
            .then(data => {
                if (data.success) {
                    // Update exact counts from server
                    document.querySelector('#voteUpBtn .count').textContent = data.counts.likes_count;
                    document.querySelector('#voteDownBtn .count').textContent = data.counts.dislikes_count;
                }
            })
            .catch(error => console.error('Error:', error));
        });
    });

    // Report Logic
    const reportBtn = document.getElementById('reportProductBtn');
    if (reportBtn) {
        reportBtn.addEventListener('click', function() {
            if (!confirm('Are you sure you want to report this product?')) return;
            
            const productId = this.getAttribute('data-id');
            const productTitle = this.getAttribute('data-title');
            const csrfMeta = document.querySelector('meta[name="csrf-token"]');
            const csrfToken = csrfMeta ? csrfMeta.getAttribute('content') : '';
            
            this.disabled = true;
            const originalHTML = this.innerHTML;
            this.innerHTML = 'Submitting...';
            
            fetch('https://creativeconor.com/report', {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/x-www-form-urlencoded',
                    'X-Requested-With': 'XMLHttpRequest'
                },
                body: `product_id=${productId}&product_title=${encodeURIComponent(productTitle)}&csrf_token=${csrfToken}`
            })
            .then(response => response.json())
            .then(data => {
